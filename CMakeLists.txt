cmake_minimum_required(VERSION 3.16)
project(SceneWall VERSION 1.7 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC ON)

# Configure OBS SDK locations (no hardcoded paths)
set(OBS_SDK_DIR "" CACHE PATH "Path to the installed OBS SDK (Windows) or obs-studio source checkout (macOS)")
if(NOT OBS_SDK_DIR)
    message(FATAL_ERROR "Set OBS_SDK_DIR to the installed OBS SDK, e.g. -DOBS_SDK_DIR=<sdk>")
endif()

set(OBS_QT_CMAKE_DIR "" CACHE PATH "Optional path to the Qt6 CMake package (lib/cmake/Qt6)")
if(OBS_QT_CMAKE_DIR)
    list(APPEND CMAKE_PREFIX_PATH "${OBS_QT_CMAKE_DIR}")
endif()

find_package(Qt6 REQUIRED COMPONENTS Widgets)

set(SCENEWALL_SOURCES
    src/plugin-main.cpp
    src/SceneWall.cpp
    src/SettingsDialog.cpp
    src/AboutDialog.h
    src/FlowLayout.h
    src/Config.h
    src/SceneContainer.h
    src/SceneTabBar.h
    src/ToggleSwitch.h
)

if(APPLE)
    # macOS: headers come from an obs-studio source checkout matching the
    # installed OBS version; binaries are linked from OBS.app itself.
    set(OBS_APP_DIR "/Applications/OBS.app" CACHE PATH "Path to OBS.app")
    set(OBS_FRAMEWORKS_DIR "${OBS_APP_DIR}/Contents/Frameworks")
    if(NOT EXISTS "${OBS_FRAMEWORKS_DIR}/libobs.framework")
        message(FATAL_ERROR "libobs.framework not found in ${OBS_FRAMEWORKS_DIR}; set -DOBS_APP_DIR=<OBS.app>")
    endif()

    set(OBS_INCLUDE_DIRS
        "${OBS_SDK_DIR}/libobs"
        "${OBS_SDK_DIR}/frontend/api"          # OBS >= 31
        "${OBS_SDK_DIR}/UI/obs-frontend-api"   # OBS <= 30
        "${OBS_SDK_DIR}/include")              # flat SDK layout

    # libobs headers need obsconfig.h (normally generated by OBS's build)
    set(OBS_GENERATED_DIR "${CMAKE_BINARY_DIR}/obs-generated")
    file(WRITE "${OBS_GENERATED_DIR}/obsconfig.h"
        "#pragma once\n#define OBS_DATA_PATH \"\"\n#define OBS_PLUGIN_PATH \"\"\n"
        "#define OBS_PLUGIN_DESTINATION \"\"\n#define OBS_RELEASE_CANDIDATE 0\n#define OBS_BETA 0\n")
    list(APPEND OBS_INCLUDE_DIRS "${OBS_GENERATED_DIR}")

    # libobs' sse-intrin.h needs SIMDe on arm64 (brew install simde)
    find_path(SIMDE_INCLUDE_DIR simde/x86/sse2.h HINTS /opt/homebrew/include /usr/local/include)
    if(NOT SIMDE_INCLUDE_DIR)
        message(FATAL_ERROR "SIMDe headers not found; run: brew install simde")
    endif()
    list(APPEND OBS_INCLUDE_DIRS "${SIMDE_INCLUDE_DIR}")

    add_library(SceneWall MODULE ${SCENEWALL_SOURCES})
    target_include_directories(SceneWall PRIVATE ${OBS_INCLUDE_DIRS})
    target_link_options(SceneWall PRIVATE "-F${OBS_FRAMEWORKS_DIR}")
    target_link_libraries(SceneWall PRIVATE
        "-framework libobs"
        "${OBS_FRAMEWORKS_DIR}/obs-frontend-api.dylib"
        Qt6::Widgets)

    set_target_properties(SceneWall PROPERTIES
        BUNDLE TRUE
        BUNDLE_EXTENSION plugin
        PREFIX ""
        OUTPUT_NAME "scene-wall"
        MACOSX_BUNDLE_GUI_IDENTIFIER "com.studiokrause.scene-wall"
        MACOSX_BUNDLE_BUNDLE_VERSION "${PROJECT_VERSION}"
        MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}"
        # Resolve libobs / Qt from OBS.app at runtime, never bundle them
        INSTALL_RPATH "@executable_path/../Frameworks"
        BUILD_WITH_INSTALL_RPATH TRUE)

    # OBS loads locale from <bundle>/Contents/Resources/locale
    file(GLOB SCENEWALL_LOCALES "${CMAKE_CURRENT_SOURCE_DIR}/data/locale/*.ini")
    target_sources(SceneWall PRIVATE ${SCENEWALL_LOCALES})
    set_source_files_properties(${SCENEWALL_LOCALES} PROPERTIES
        MACOSX_PACKAGE_LOCATION "Resources/locale")

    # Ad-hoc sign so Gatekeeper / hardened OBS accepts the bundle
    add_custom_command(TARGET SceneWall POST_BUILD
        COMMAND codesign --force --deep --sign - "$<TARGET_BUNDLE_DIR:SceneWall>"
        VERBATIM)

    install(TARGETS SceneWall LIBRARY DESTINATION ".")
else()
    include_directories("${OBS_SDK_DIR}/include")
    link_directories("${OBS_SDK_DIR}/lib")

    add_library(SceneWall MODULE ${SCENEWALL_SOURCES})
    target_link_libraries(SceneWall PRIVATE obs.lib obs-frontend-api.lib Qt6::Widgets)
    set_target_properties(SceneWall PROPERTIES PREFIX "" OUTPUT_NAME "scene-wall")

    install(TARGETS SceneWall
        RUNTIME DESTINATION "plugins/scene-wall"
        LIBRARY DESTINATION "plugins/scene-wall")
    install(DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/data/locale"
        DESTINATION "plugins/scene-wall/data")
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(ConfigTest tests/ConfigTest.cpp)
    target_link_libraries(ConfigTest PRIVATE Qt6::Gui)
    add_test(NAME ConfigTest COMMAND ConfigTest)
    add_executable(SchedulerTest tests/SchedulerTest.cpp)
    if(APPLE)
        target_include_directories(SchedulerTest PRIVATE src ${OBS_INCLUDE_DIRS})
    else()
        target_include_directories(SchedulerTest PRIVATE src "${OBS_SDK_DIR}/include")
    endif()
    target_link_libraries(SchedulerTest PRIVATE Qt6::Widgets)
    add_test(NAME SchedulerTest COMMAND SchedulerTest)
endif()
